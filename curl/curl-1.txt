# 1. Successful upload of a PNG or JPEG (returns 202 Accepted + Location header)
curl -i -X POST http://localhost:4000/v1/images \
  -F "image=@frontend/favicon.ico" # or path to an image

# 2. Trigger validation failure: Unsupported file type (e.g. text/plain or markdown)
curl -i -X POST http://localhost:4000/v1/images \
  -F "image=@README.md"

# 3. Trigger validation failure: Missing 'image' form field
curl -i -X POST http://localhost:4000/v1/images \
  -F "wrong_field=@README.md"

# 4. Check job status by public ID
curl -i http://localhost:4000/v1/jobs/REPLACE_WITH_JOB_ID

# 5. Query nonexistent job (returns 404 Not Found)
curl -i http://localhost:4000/v1/jobs/00000000-0000-0000-0000-000000000000
